    𑌗𑍁𑌰𑍁𑌪𑍍𑌰𑌕𑌮𑍍𑌪𑌨𑌸𑍍𑌥𑌲𑍇𑌶𑌪𑌞𑍍𑌚𑌰𑌤𑍍𑌨𑌬𑍋𑌧𑌿𑌕𑌾𑌂 𑌸𑌦𑌾𑌕𑌲𑌯𑍍𑌯 𑌮𑌾𑌤𑍃𑌮𑌾𑌤𑍃𑌮𑌾𑌤𑌰𑌂 𑌚 𑌮𑌾𑌨𑌸𑍇 ।
    𑌅𑌨𑌨𑍍𑌤𑌰𑌾𑌮𑌦𑍀𑌕𑍍𑌷𑌿𑌤𑌂 𑌮𑌹𑍇𑌶𑌕𑌾𑌲𑌿𑌕𑌾𑌪𑍁𑌰𑍇 𑌭𑌵𑌂 𑌸𑍁𑌲𑍇𑌖𑌕𑌂 𑌪𑍍𑌰𑌣𑌮𑍍𑌯 𑌭𑌾𑌷𑍍𑌯𑌯𑌤𑍍𑌨𑌮𑌾𑌰𑌭𑍇 ॥

𑌶𑌿𑌵𑌕𑌾𑌲𑍀𑌪𑍁𑌰𑌸𑍍𑌯 𑌅𑌨𑌨𑍍𑌤𑌰𑌾𑌮𑌦𑍀𑌕𑍍𑌷𑌿𑌤𑌨𑌾𑌮𑌾 𑌸𑍁𑌵𑌿𑌖𑍍𑌯𑌾𑌤𑌃 𑌪𑍍𑌰𑌵𑌕𑍍𑌤𑌾 𑌶𑍍𑌰𑍌𑌤𑍀 𑌚 𑌵𑍍𑌯𑌰𑌾𑌜𑌤 । 𑌗𑍁𑌰𑍁𑌵𑌾𑌯𑍁𑌪𑍁𑌰𑌾𑌧𑍀𑌶𑍇 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑍇 𑌅𑌪𑌾𑌰𑌭𑌕𑍍𑌤𑌿𑌮𑌾𑌁𑌶𑍍𑌚 । 𑌗𑍁𑌰𑍁𑌵𑌾𑌤𑌪𑍁𑌰𑍀𑌶𑌪𑌞𑍍𑌚𑌰𑌤𑍍𑌨𑌨𑌾𑌮𑌿𑌕𑌾𑌮𑍁𑌪𑌜𑌾𑌤𑌿𑌵𑍃𑌤𑍍𑌤𑌨𑌿𑌬𑌦𑍍𑌧𑌾𑌂 𑌤𑌤𑍍𑌸𑍍𑌤𑍁𑌤𑌿𑌂 𑌵𑍍𑌯𑌾𑌖𑍍𑌯𑌾𑌸𑍍𑌯𑌾𑌮𑌃 ।

    𑌕𑌲𑍍𑌯𑌾𑌣𑌰𑍂𑌪𑌾𑌯 𑌕𑌲𑍌 𑌜𑌨𑌾𑌨𑌾𑌂 𑌕𑌲𑍍𑌯𑌾𑌣𑌦𑌾𑌤𑍍𑌰𑍇 𑌕𑌰𑍁𑌣𑌾𑌸𑍁𑌧𑌾𑌬𑍍𑌧𑍇 ।
    𑌕𑌮𑍍𑌬𑍍𑌵𑌾𑌦𑌿𑌦𑌿𑌵𑍍𑌯𑌾𑌯𑍁𑌧𑌸𑌤𑍍𑌕𑌰𑌾𑌯 𑌵𑌾𑌤𑌾𑌲𑌯𑌾𑌧𑍀𑌶 𑌨𑌮𑍋 𑌨𑌮𑌸𑍍𑌤𑍇 ॥ ௧ ॥

𑌕𑌲𑍍𑌯𑌾𑌣𑌰𑍂𑌪𑌾𑌯 । 𑌕𑌲𑍍𑌯𑌾𑌣𑌂 𑌕𑌲𑍍𑌯𑍇 𑌪𑍍𑌰𑌾𑌤𑌃 𑌅𑌣𑍍𑌯𑌤𑍇 𑌶𑌬𑍍𑌦𑍍𑌯𑌤𑍇 𑌇𑌤𑌿 𑌮𑌙𑍍𑌗𑌳𑌮𑍍 । 𑌤𑌥𑌾 𑌰𑍂𑌪𑌂 𑌯𑌸𑍍𑌯 𑌤𑌥𑌾𑌭𑍂𑌤𑌸𑍍𑌤𑌸𑍍𑌮𑍈 । 𑌨𑌮𑍋 𑌨𑌮 𑌇𑌤𑍍𑌯𑌨𑍍𑌵𑌯𑌃 । 𑌨𑌮𑌃𑌸𑍍𑌵𑌸𑍍𑌤𑌿𑌸𑍍𑌵𑌾𑌹𑌾𑌸𑍍𑌵𑌧𑌾𑌲𑌮𑍍𑌵𑌷𑌡𑍍𑌯𑍋𑌗𑌾𑌚𑍍𑌚𑍇𑌤𑌿 𑌚𑌤𑍁𑌰𑍍𑌥𑍀 𑌕𑌰𑍍𑌤𑍁𑌰𑍀𑌪𑍍𑌸𑌿𑌤𑌤𑌮𑌸𑍍𑌯𑌾𑌪𑌿 𑌭𑌗𑌵𑌤𑌃 । 𑌕𑌵𑌿𑌃 𑌸𑍍𑌤𑍋𑌤𑍍𑌰𑌮𑍁𑌪𑌕𑍍𑌰𑌮𑌤𑍇 𑌮𑌙𑍍𑌗𑌳𑌵𑌾𑌚𑌕𑍇𑌨 𑌕𑌲𑍍𑌯𑌾𑌣𑌶𑌬𑍍𑌦𑍇𑌨 । 𑌕𑌲𑍌 𑌤𑌨𑍍𑌨𑌾𑌮𑌕𑍇 𑌵𑌰𑍍𑌤𑌮𑌾𑌨𑍇 𑌯𑍁𑌗𑍇 । 𑌕𑌲𑍍𑌯𑌤𑍇 𑌪𑌾𑌪𑍇𑌷𑍁 𑌨𑌿𑌕𑍍𑌷𑌿𑌪𑍍𑌯𑌤𑍇𑌨𑍇𑌨 𑌯𑌦𑍍𑌵𑌾 𑌕𑌲𑌯𑌤𑌿 𑌪𑌾𑌪𑍇𑌨 𑌜𑌡𑌯𑌤𑌿 𑌕𑌲𑍁𑌷𑌿𑌤𑌂 𑌕𑌰𑍋𑌤𑍀𑌤𑌿 𑌕𑌲𑌿𑌃 । 𑌜𑌨𑌾𑌨𑌾𑌂 𑌜𑌾𑌯𑌤 𑌇𑌤𑌿 𑌜𑌨𑌸𑍍𑌤𑍇𑌷𑌾𑌂 𑌮𑌰𑍍𑌤𑍍𑌯𑌲𑍋𑌕𑌾𑌨𑌾𑌮𑍍 । 𑌕𑌲𑍌 𑌯𑍁𑌗𑍇 𑌯𑌜𑍍𑌞𑌤𑌪𑌆𑌦𑌿𑌭𑍍𑌯𑍋 𑌅𑌰𑍍𑌚𑌾𑌮𑍂𑌰𑍍𑌤𑌿𑌸𑍇𑌵𑌾𑌯𑌾 𑌉𑌚𑌿𑌤𑌤𑌰𑌤𑍍𑌵𑌾𑌤𑍍 𑌤𑌦𑍍𑌯𑍁𑌗𑍇 𑌜𑌨𑌾𑌨𑌾𑌂 𑌵𑌿𑌶𑌿𑌷𑍍𑌟𑌃 𑌕𑌲𑍍𑌯𑌾𑌣𑌰𑍂𑌪 𑌇𑌤𑌿 𑌯𑌾𑌵𑌤𑍍 । 𑌅𑌥 𑌕𑌲𑍍𑌯𑌾𑌣𑌸𑍍𑌯 𑌸𑌦𑍍𑌗𑌤𑍇𑌃 𑌦𑌾𑌤𑍍𑌰𑍇 𑌪𑍍𑌰𑌦𑌾𑌯𑌕𑌸𑍍𑌤𑌸𑍍𑌮𑍈 𑌨𑌮𑍋 𑌨𑌮 𑌇𑌤𑍍𑌯𑌤𑍍𑌰 𑌪𑍁𑌨𑌰𑌨𑍍𑌵𑌯𑌃 । 𑌕𑌰𑍁𑌣𑌾 𑌦𑌯𑍈𑌵 𑌸𑍁𑌧𑌾𑌮𑍃𑌤𑌮𑌂 𑌕𑌰𑍁𑌣𑌾𑌸𑍁𑌧𑌾 । 𑌤𑌸𑍍𑌯𑌾 𑌅𑌬𑍍𑌧𑌿𑌃 𑌕𑌰𑍍𑌮𑌣𑍍𑌯𑌧𑌿𑌕𑌰𑌣𑍇 𑌚𑍇𑌤𑌿 𑌸𑌾𑌗𑌰𑌃 𑌦𑌯𑌾𑌮𑍃𑌤𑌸𑌾𑌗𑌰𑌃 𑌸𑌮𑍍𑌬𑍁𑌦𑍍𑌧𑍌 । 𑌕𑌮𑍍𑌬𑍁𑌃 𑌶𑌙𑍍𑌖𑍋 𑌨𑌿𑌪𑌾𑌤𑍍𑌯𑌤𑍇 । 𑌸 𑌪𑌾𑌞𑍍𑌚𑌜𑌨𑍍𑌯𑌨𑌾𑌮𑌾 𑌆𑌦𑌿𑌰𑍍𑌯𑍇𑌷𑌾𑌂 𑌤𑍇 𑌦𑌿𑌵𑍍𑌯𑌾𑌯𑍁𑌧𑌾𑌨𑌿 𑌸𑍁𑌦𑌰𑍍𑌶𑌨𑌕𑍌𑌮𑍋𑌦𑌕𑍀𑌨𑌨𑍍𑌦𑌕𑌶𑌾𑌰𑍍𑌙𑍍𑌗𑌾𑌦𑍀𑌨𑌿 । 𑌸𑌤𑍍𑌕𑌰𑌾𑌃 𑌸𑌾𑌧𑍁𑌹𑌸𑍍𑌤𑌾𑌃 । 𑌕𑌮𑍍𑌬𑍍𑌵𑌾𑌦𑌿𑌦𑌿𑌵𑍍𑌯𑌾𑌯𑍁𑌧𑌾𑌨𑌿 𑌸𑌤𑍍𑌕𑌰𑍇𑌷𑍁 𑌯𑌸𑍍𑌯 𑌤𑌸𑍍𑌮𑍈 𑌨𑌮𑍋 𑌨𑌮 𑌇𑌤𑍍𑌯𑌤𑍍𑌰𑍈𑌵 𑌪𑍁𑌨𑌰𑌨𑍍𑌵𑌯𑌃 । 𑌵𑌾𑌤𑌃 𑌵𑌾𑌤𑍀𑌤𑌿 𑌕𑌰𑍍𑌤𑌰𑌿 𑌕𑍍𑌤𑌾𑌨𑍍𑌤𑌃 𑌪𑌞𑍍𑌚𑌭𑍂𑌤𑌾𑌨𑍍𑌤𑌰𑍍𑌮𑌤𑌚𑌤𑍁𑌰𑍍𑌥𑌭𑍂𑌤𑍋 𑌵𑌾𑌯𑍁𑌃 । 𑌤𑌸𑍍𑌯 𑌤𑍇𑌨 𑌨𑌿𑌰𑍍𑌮𑌿𑌤𑌃 𑌆𑌲𑌯𑌃 𑌵𑌾𑌸𑌸𑍍𑌥𑌾𑌨𑌂 𑌤𑌸𑍍𑌯𑌾𑌧𑍀𑌶𑌃 𑌸𑍍𑌵𑌾𑌮𑍀 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑌃 𑌸𑌮𑍍𑌬𑍁𑌦𑍍𑌧𑍌 । 𑌨𑌮𑍋 𑌨𑌮 𑌇𑌤𑍍𑌯𑌾𑌮𑍍𑌰𑍇𑌡𑌿𑌤𑌮𑍍 । 𑌤𑍇 𑌤𑍁𑌭𑍍𑌯𑌂 𑌨𑌮𑌃𑌸𑍍𑌵𑌸𑍍𑌤𑌿𑌸𑍍𑌵𑌾𑌹𑌾𑌸𑍍𑌵𑌧𑌾𑌲𑌮𑍍𑌵𑌷𑌡𑍍𑌯𑍋𑌗𑌾𑌚𑍍𑌚𑍇𑌤𑌿 𑌪𑍁𑌨𑌶𑍍𑌚𑌤𑍁𑌰𑍍𑌥𑍀 ।

    𑌨𑌾𑌰𑌾𑌯𑌣𑍇𑌤𑍍𑌯𑌾𑌦𑌿𑌜𑌪𑌦𑍍𑌭𑌿𑌰𑍁𑌚𑍍𑌚𑍈𑌰𑍍𑌭𑌕𑍍𑌤𑍈𑌃 𑌸𑌦𑌾𑌪𑍂𑌰𑍍𑌣𑌮𑌹𑌾𑌲𑌯𑌾𑌯 ।
    𑌸𑍍𑌵𑌤𑍀𑌰𑍍𑌥𑌗𑌾𑌙𑍍𑌗𑍋𑌪𑌮𑌵𑌾𑌰𑌿𑌮𑌗𑍍𑌨𑌨𑌿𑌵𑌰𑍍𑌤𑌿𑌤𑌾𑌶𑍇𑌷𑌰𑍁𑌜𑍇 𑌨𑌮𑌸𑍍𑌤𑍇 ॥ ௨ ॥

𑌨𑌾𑌰𑌾𑌯𑌣 𑌭𑌗𑌵𑌨𑍍𑌨𑌾𑌮 𑌸𑌮𑍍𑌬𑍁𑌦𑍍𑌧𑍌 । 𑌇𑌤𑍍𑌯𑌾𑌦𑌿 𑌨 𑌕𑍇𑌵𑌲𑌂 𑌨𑌾𑌰𑌾𑌯𑌣𑍇𑌤𑌿 𑌪𑍍𑌰𑌤𑍍𑌯𑍁𑌤 𑌰𑌾𑌮𑌕𑍃𑌷𑍍𑌣𑌹𑌰𑌿𑌨𑍃𑌹𑌰𑍀𑌤𑌿 𑌅𑌚𑍍𑌯𑍁𑌤𑌾𑌨𑌨𑍍𑌤𑌗𑍋𑌵𑌿𑌨𑍍𑌦𑍇𑌤𑌿 𑌕𑍇𑌶𑌵𑌮𑌾𑌧𑌵𑌦𑌾𑌮𑍋𑌦𑌰𑍇𑌤𑌿 𑌵𑌾 ।

𑌇𑌤𑍍𑌥𑌂 𑌤𑍁 𑌸𑍍𑌤𑍋𑌤𑍍𑌰𑌪𑌾𑌰𑌾𑌯𑌣𑌾𑌵𑌸𑌰𑍇 𑌪𑍍𑌰𑌤𑍍𑌯𑍇𑌕𑌂 𑌶𑍍𑌲𑍋𑌕𑌸𑍍𑌯 𑌪𑌶𑍍𑌚𑌾𑌜𑍍𑌜𑌪𑍍𑌯𑌤𑍇 । 𑌧𑍍𑌯𑌾𑌯𑌨𑍍𑌕𑍃𑌤𑍇 𑌯𑌜𑌨𑍍𑌯𑌜𑍍𑌞𑍈𑌸𑍍𑌤𑍍𑌰𑍇𑌤𑌾𑌯𑌾𑌂 𑌦𑍍𑌵𑌾𑌪𑌰𑍇𑌰𑍍𑌚𑌯𑌨𑍍 । 𑌯𑌦𑌾𑌪𑍍𑌨𑍋𑌤𑌿 𑌤𑌦𑌾𑌪𑍍𑌨𑍋𑌤𑌿 𑌕𑌲𑍌 𑌸𑌙𑍍𑌕𑍀𑌰𑍍𑌤𑍍𑌯 𑌕𑍇𑌶𑌵𑌮𑍍 ॥ 𑌇𑌤𑌿 𑌪𑍍𑌰𑌸𑌿𑌦𑍍𑌧𑌾𑌯𑌾𑌃 𑌶𑍍𑌰𑍀𑌮𑌦𑍍𑌭𑌾𑌗𑌵𑌤𑍋𑌕𑍍𑌤𑍇𑌰𑌨𑍁𑌗𑍁𑌣𑌂 𑌨𑌾𑌮𑌜𑌪𑌮𑌨𑍁𑌸𑌰𑍍𑌯𑌤𑍇 । 𑌨𑌾𑌰𑌾𑌯𑌣𑍇𑌤𑌿 𑌭𑌗𑌵𑌨𑍍𑌨𑌾𑌮 𑌨𑌾𑌨𑌾𑌰𑍍𑌥𑌮𑌤𑍍 । 𑌨𑌾𑌰𑌾 𑌜𑌲𑌮𑍍 𑌅𑌯𑌨𑌂 𑌸𑍍𑌥𑌾𑌨𑌂 𑌯𑌸𑍍𑌯𑍇𑌤𑌿 । 𑌯𑌦𑍍𑌵𑌾 𑌸𑌰𑍍𑌵𑍇 𑌗𑌤𑍍𑌯𑌰𑍍𑌥𑌾𑌃 𑌪𑍍𑌰𑌾𑌪𑍍𑌤𑍍𑌯𑌰𑍍𑌥𑌾 𑌇𑌤𑌿 𑌨𑌾𑌰𑌸𑍍𑌯 𑌜𑍍𑌞𑌾𑌨𑌸𑍍𑌯 𑌮𑍁𑌕𑍍𑌤𑍇𑌰𑍍𑌵𑌾 𑌅𑌯𑌨𑌂 𑌪𑍍𑌰𑌾𑌪𑍍𑌤𑌿𑌰𑍍𑌯𑌸𑍍𑌮𑌾𑌦𑌿𑌤𑌿 । 𑌯𑌦𑍍𑌵𑌾 𑌨𑌰𑌾𑌣𑌾𑌂 𑌸𑌮𑍂𑌹𑍋 𑌨𑌾𑌰𑌂 𑌤𑌤𑍍𑌰𑌾𑌯𑌨𑌂 𑌸𑍍𑌥𑌾𑌨𑌂 𑌯𑌸𑍍𑌯𑍇𑌤𑌿 । 𑌇𑌤𑌿 𑌶𑌤𑌶𑌃 𑌵𑍍𑌯𑍁𑌤𑍍𑌪𑌤𑍍𑌤𑌯𑌃 ।
