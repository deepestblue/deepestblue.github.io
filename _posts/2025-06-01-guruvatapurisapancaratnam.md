    𑌗𑍁𑌰𑍁𑌪𑍍𑌰𑌕𑌮𑍍𑌪𑌨𑌸𑍍𑌥𑌲𑍇𑌶𑌪𑌞𑍍𑌚𑌰𑌤𑍍𑌨𑌬𑍋𑌧𑌿𑌕𑌾𑌂 𑌸𑌦𑌾𑌕𑌲𑌯𑍍𑌯 𑌮𑌾𑌤𑍃𑌮𑌾𑌤𑍃𑌮𑌾𑌤𑌰𑌂 𑌚 𑌮𑌾𑌨𑌸𑍇 ।
    𑌅𑌨𑌨𑍍𑌤𑌰𑌾𑌮𑌦𑍀𑌕𑍍𑌷𑌿𑌤𑌂 𑌮𑌹𑍇𑌶𑌕𑌾𑌲𑌿𑌕𑌾𑌪𑍁𑌰𑍇 𑌭𑌵𑌂 𑌸𑍁𑌲𑍇𑌖𑌕𑌂 𑌪𑍍𑌰𑌣𑌮𑍍𑌯 𑌭𑌾𑌷𑍍𑌯𑌯𑌤𑍍𑌨𑌮𑌾𑌰𑌭𑍇 ॥

𑌶𑌿𑌵𑌕𑌾𑌲𑍀𑌪𑍁𑌰𑌸𑍍𑌯 𑌅𑌨𑌨𑍍𑌤𑌰𑌾𑌮𑌦𑍀𑌕𑍍𑌷𑌿𑌤𑌨𑌾𑌮𑌾 𑌸𑍁𑌵𑌿𑌖𑍍𑌯𑌾𑌤𑌃 𑌪𑍍𑌰𑌵𑌕𑍍𑌤𑌾 𑌶𑍍𑌰𑍌𑌤𑍀 𑌚 𑌵𑍍𑌯𑌰𑌾𑌜𑌤 । 𑌗𑍁𑌰𑍁𑌵𑌾𑌯𑍁𑌪𑍁𑌰𑌾𑌧𑍀𑌶𑍇 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑍇 𑌅𑌪𑌾𑌰𑌭𑌕𑍍𑌤𑌿𑌮𑌾𑌁𑌶𑍍𑌚 । 𑌗𑍁𑌰𑍁𑌵𑌾𑌤𑌪𑍁𑌰𑍀𑌶𑌪𑌞𑍍𑌚𑌰𑌤𑍍𑌨𑌨𑌾𑌮𑌿𑌕𑌾𑌮𑍁𑌪𑌜𑌾𑌤𑌿𑌵𑍃𑌤𑍍𑌤𑌨𑌿𑌬𑌦𑍍𑌧𑌾𑌂 𑌤𑌤𑍍𑌸𑍍𑌤𑍁𑌤𑌿𑌂 𑌵𑍍𑌯𑌾𑌖𑍍𑌯𑌾𑌸𑍍𑌯𑌾𑌮𑌃 ।

    𑌕𑌲𑍍𑌯𑌾𑌣𑌰𑍂𑌪𑌾𑌯 𑌕𑌲𑍌 𑌜𑌨𑌾𑌨𑌾𑌂 𑌕𑌲𑍍𑌯𑌾𑌣𑌦𑌾𑌤𑍍𑌰𑍇 𑌕𑌰𑍁𑌣𑌾𑌸𑍁𑌧𑌾𑌬𑍍𑌧𑍇 ।
    𑌕𑌮𑍍𑌬𑍍𑌵𑌾𑌦𑌿𑌦𑌿𑌵𑍍𑌯𑌾𑌯𑍁𑌧𑌸𑌤𑍍𑌕𑌰𑌾𑌯 𑌵𑌾𑌤𑌾𑌲𑌯𑌾𑌧𑍀𑌶 𑌨𑌮𑍋 𑌨𑌮𑌸𑍍𑌤𑍇 ॥ ௧ ॥

𑌕𑌲𑍍𑌯𑌾𑌣𑌰𑍂𑌪𑌾𑌯 । 𑌕𑌲𑍍𑌯𑌾𑌣𑌂 𑌕𑌲𑍍𑌯𑍇 𑌪𑍍𑌰𑌾𑌤𑌰𑍍𑌣𑍍𑌯𑌤𑍇 𑌶𑌬𑍍𑌦𑍍𑌯𑌤𑍇 𑌇𑌤𑌿 𑌮𑌙𑍍𑌗𑌳𑌮𑍍 । 𑌤𑌥𑌾 𑌰𑍂𑌪𑌂 𑌯𑌸𑍍𑌯 𑌤𑌥𑌾𑌭𑍂𑌤𑌸𑍍𑌤𑌸𑍍𑌮𑍈 । 𑌨𑌮𑍋 𑌨𑌮 𑌇𑌤𑍍𑌯𑌨𑍍𑌵𑌯𑌃 । 𑌨𑌮𑌃𑌸𑍍𑌵𑌸𑍍𑌤𑌿𑌸𑍍𑌵𑌾𑌹𑌾𑌸𑍍𑌵𑌧𑌾𑌲𑌮𑍍𑌵𑌷𑌡𑍍𑌯𑍋𑌗𑌾𑌚𑍍𑌚𑍇𑌤𑌿 𑌚𑌤𑍁𑌰𑍍𑌥𑍀 𑌕𑌰𑍍𑌤𑍁𑌰𑍀𑌪𑍍𑌸𑌿𑌤𑌤𑌮𑌸𑍍𑌯𑌾𑌪𑌿 𑌭𑌗𑌵𑌤𑌃 । 𑌕𑌵𑌿𑌃 𑌸𑍍𑌤𑍋𑌤𑍍𑌰𑌮𑍁𑌪𑌕𑍍𑌰𑌮𑌤𑍇 𑌮𑌙𑍍𑌗𑌳𑌵𑌾𑌚𑌕𑍇𑌨 𑌕𑌲𑍍𑌯𑌾𑌣𑌶𑌬𑍍𑌦𑍇𑌨 । 𑌕𑌲𑍌 𑌤𑌨𑍍𑌨𑌾𑌮𑌕𑍇 𑌵𑌰𑍍𑌤𑌮𑌾𑌨𑍇 𑌯𑍁𑌗𑍇 । 𑌕𑌲𑍍𑌯𑌤𑍇 𑌪𑌾𑌪𑍇𑌷𑍁 𑌨𑌿𑌕𑍍𑌷𑌿𑌪𑍍𑌯𑌤𑍇𑌨𑍇𑌨 𑌯𑌦𑍍𑌵𑌾 𑌕𑌲𑌯𑌤𑌿 𑌪𑌾𑌪𑍇𑌨 𑌜𑌡𑌯𑌤𑌿 𑌕𑌲𑍁𑌷𑌿𑌤𑌂 𑌕𑌰𑍋𑌤𑍀𑌤𑌿 𑌕𑌲𑌿𑌃 । 𑌜𑌨𑌾𑌨𑌾𑌂 𑌜𑌾𑌯𑌤 𑌇𑌤𑌿 𑌜𑌨𑌸𑍍𑌤𑍇𑌷𑌾𑌂 𑌮𑌰𑍍𑌤𑍍𑌯𑌲𑍋𑌕𑌾𑌨𑌾𑌮𑍍 । 𑌕𑌲𑍌 𑌯𑍁𑌗𑍇 𑌯𑌜𑍍𑌞𑌤𑌪𑌆𑌦𑌿𑌭𑍍𑌯𑍋 𑌅𑌰𑍍𑌚𑌾𑌮𑍂𑌰𑍍𑌤𑌿𑌸𑍇𑌵𑌾𑌯𑌾 𑌉𑌚𑌿𑌤𑌤𑌰𑌤𑍍𑌵𑌾𑌤𑍍 𑌤𑌦𑍍𑌯𑍁𑌗𑍇 𑌜𑌨𑌾𑌨𑌾𑌂 𑌵𑌿𑌶𑌿𑌷𑍍𑌟𑌃 𑌕𑌲𑍍𑌯𑌾𑌣𑌰𑍂𑌪 𑌇𑌤𑌿 𑌯𑌾𑌵𑌤𑍍 । 𑌅𑌥 𑌕𑌲𑍍𑌯𑌾𑌣𑌸𑍍𑌯 𑌸𑌦𑍍𑌗𑌤𑍇𑌃 𑌦𑌾𑌤𑍍𑌰𑍇 𑌪𑍍𑌰𑌦𑌾𑌯𑌕𑌸𑍍𑌤𑌸𑍍𑌮𑍈 𑌨𑌮𑍋 𑌨𑌮 𑌇𑌤𑍍𑌯𑌤𑍍𑌰 𑌪𑍁𑌨𑌰𑌨𑍍𑌵𑌯𑌃 । 𑌕𑌰𑍁𑌣𑌾 𑌦𑌯𑍈𑌵 𑌸𑍁𑌧𑌾𑌮𑍃𑌤𑌮𑌂 𑌕𑌰𑍁𑌣𑌾𑌸𑍁𑌧𑌾 । 𑌤𑌸𑍍𑌯𑌾 𑌅𑌬𑍍𑌧𑌿𑌃 𑌕𑌰𑍍𑌮𑌣𑍍𑌯𑌧𑌿𑌕𑌰𑌣𑍇 𑌚𑍇𑌤𑌿 𑌸𑌾𑌗𑌰𑌃 𑌦𑌯𑌾𑌮𑍃𑌤𑌸𑌾𑌗𑌰𑌃 𑌸𑌮𑍍𑌬𑍁𑌦𑍍𑌧𑍌 । 𑌕𑌮𑍍𑌬𑍁𑌃 𑌶𑌙𑍍𑌖𑍋 𑌨𑌿𑌪𑌾𑌤𑍍𑌯𑌤𑍇 । 𑌸 𑌪𑌾𑌞𑍍𑌚𑌜𑌨𑍍𑌯𑌨𑌾𑌮𑌾 𑌆𑌦𑌿𑌰𑍍𑌯𑍇𑌷𑌾𑌂 𑌤𑍇 𑌦𑌿𑌵𑍍𑌯𑌾𑌯𑍁𑌧𑌾𑌨𑌿 𑌸𑍁𑌦𑌰𑍍𑌶𑌨𑌕𑍌𑌮𑍋𑌦𑌕𑍀𑌨𑌨𑍍𑌦𑌕𑌶𑌾𑌰𑍍𑌙𑍍𑌗𑌾𑌦𑍀𑌨𑌿 । 𑌸𑌤𑍍𑌕𑌰𑌾𑌃 𑌸𑌾𑌧𑍁𑌹𑌸𑍍𑌤𑌾𑌃 । 𑌕𑌮𑍍𑌬𑍍𑌵𑌾𑌦𑌿𑌦𑌿𑌵𑍍𑌯𑌾𑌯𑍁𑌧𑌾𑌨𑌿 𑌸𑌤𑍍𑌕𑌰𑍇𑌷𑍁 𑌯𑌸𑍍𑌯 𑌤𑌸𑍍𑌮𑍈 𑌨𑌮𑍋 𑌨𑌮 𑌇𑌤𑍍𑌯𑌤𑍍𑌰𑍈𑌵 𑌪𑍁𑌨𑌰𑌨𑍍𑌵𑌯𑌃 । 𑌵𑌾𑌤𑌃 𑌵𑌾𑌤𑍀𑌤𑌿 𑌕𑌰𑍍𑌤𑌰𑌿 𑌕𑍍𑌤𑌾𑌨𑍍𑌤𑌃 𑌪𑌞𑍍𑌚𑌭𑍂𑌤𑌾𑌨𑍍𑌤𑌰𑍍𑌮𑌤𑌚𑌤𑍁𑌰𑍍𑌥𑌭𑍂𑌤𑍋 𑌵𑌾𑌯𑍁𑌃 । 𑌤𑌸𑍍𑌯 𑌤𑍇𑌨 𑌨𑌿𑌰𑍍𑌮𑌿𑌤𑌃 𑌆𑌲𑌯𑌃 𑌵𑌾𑌸𑌸𑍍𑌥𑌾𑌨𑌂 𑌤𑌸𑍍𑌯𑌾𑌧𑍀𑌶𑌃 𑌸𑍍𑌵𑌾𑌮𑍀 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑌃 𑌸𑌮𑍍𑌬𑍁𑌦𑍍𑌧𑍌 । 𑌨𑌮𑍋 𑌨𑌮 𑌇𑌤𑍍𑌯𑌾𑌮𑍍𑌰𑍇𑌡𑌿𑌤𑌮𑍍 । 𑌤𑍇 𑌤𑍁𑌭𑍍𑌯𑌂 𑌨𑌮𑌃𑌸𑍍𑌵𑌸𑍍𑌤𑌿𑌸𑍍𑌵𑌾𑌹𑌾𑌸𑍍𑌵𑌧𑌾𑌲𑌮𑍍𑌵𑌷𑌡𑍍𑌯𑍋𑌗𑌾𑌚𑍍𑌚𑍇𑌤𑌿 𑌪𑍁𑌨𑌶𑍍𑌚𑌤𑍁𑌰𑍍𑌥𑍀 । 𑌪𑍍𑌰𑌥𑌮𑌾𑌕𑍍𑌷𑌰𑌪𑍍𑌰𑌾𑌸𑍇𑌨 𑌵𑌿𑌶𑍇𑌷𑌃 ।

    𑌨𑌾𑌰𑌾𑌯𑌣𑍇𑌤𑍍𑌯𑌾𑌦𑌿 𑌜𑌪𑌦𑍍𑌭𑌿𑌰𑍁𑌚𑍍𑌚𑍈𑌰𑍍𑌭𑌕𑍍𑌤𑍈𑌃 𑌸𑌦𑌾 𑌪𑍂𑌰𑍍𑌣𑌮𑌹𑌾𑌲𑌯𑌾𑌯 ।
    𑌸𑍍𑌵𑌤𑍀𑌰𑍍𑌥𑌗𑌾𑌙𑍍𑌗𑍋𑌪𑌮𑌵𑌾𑌰𑌿𑌮𑌗𑍍𑌨 𑌨𑌿𑌵𑌰𑍍𑌤𑌿𑌤𑌾𑌶𑍇𑌷𑌰𑍁𑌜𑍇 𑌨𑌮𑌸𑍍𑌤𑍇 ॥ ௨ ॥

    𑌬𑍍𑌰𑌾𑌹𑍍𑌮𑍇 𑌮𑍁𑌹𑍂𑌰𑍍𑌤𑍇 𑌪𑌰𑌿𑌤𑌃 𑌸𑍍𑌵𑌭𑌕𑍍𑌤𑍈𑌃 𑌸𑌨𑍍𑌦𑍃𑌷𑍍𑌟𑌸𑌰𑍍𑌵𑍋𑌤𑍍𑌤𑌮𑌵𑌿𑌶𑍍𑌵𑌰𑍂𑌪 ।
    𑌸𑍍𑌵𑌤𑍈𑌲𑌸𑌂𑌸𑍇𑌵𑌕𑌰𑍋𑌗𑌹𑌰𑍍𑌤𑍍𑌰𑍇 𑌵𑌾𑌤𑌾𑌲𑌯𑌾𑌧𑍀𑌶 𑌨𑌮𑍋 𑌨𑌮𑌸𑍍𑌤𑍇 ॥ ௩ ॥

𑌬𑍍𑌰𑌾𑌹𑍍𑌮𑍇 𑌬𑍍𑌰𑌹𑍍𑌮𑌣𑍋𑌯𑌂 𑌬𑍍𑌰𑌾𑌹𑍍𑌮𑌸𑍍𑌤𑌸𑍍𑌮𑌿𑌨𑍍 । 𑌮𑍁𑌹𑍂𑌰𑍍𑌤𑍇 𑌕𑌾𑌲𑌪𑌰𑌿𑌮𑌾𑌣𑌵𑌿𑌶𑍇𑌷𑌸𑍍𑌤𑌸𑍍𑌮𑌿𑌨𑍍 । 𑌬𑍍𑌰𑌾𑌹𑍍𑌮𑌮𑍁𑌹𑍂𑌰𑍍𑌤𑌸𑍍𑌤𑍂𑌷𑌸𑌃 𑌪𑍍𑌰𑌾𑌕𑍍𑌪𑌰𑌿𑌗𑌣𑌿𑌤𑌃 𑌤𑌸𑍍𑌮𑌿𑌂𑌸𑍍𑌤𑌦𑌾 । 𑌪𑌰𑌿𑌤𑌃 𑌸𑌰𑍍𑌵𑌤𑌃 । 𑌸𑍍𑌵𑌭𑌕𑍍𑌤𑍈𑌃 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑌸𑍇𑌵𑌕𑍈𑌃 । 𑌸𑌨𑍍𑌦𑍃𑌷𑍍𑌟𑌸𑌰𑍍𑌬𑍋𑌤𑍍𑌤𑌮𑌵𑌿𑌶𑍍𑌵𑌰𑍂𑌪 𑌸𑌨𑍍𑌦𑍃𑌷𑍍𑌟𑌂 𑌸𑌮𑍍𑌯𑌗𑍍𑌵𑍀𑌕𑍍𑌷𑌿𑌤𑌂 𑌸𑍍𑌵𑌭𑌕𑍍𑌤𑍈𑌰𑌿𑌤𑌿 𑌯𑌾𑌵𑌤𑍍 । 𑌸𑌰𑍍𑌵𑍋𑌤𑍍𑌤𑌮𑌵𑌿𑌶𑍍𑌵𑌰𑍂𑌪𑌂 𑌸𑌰𑍍𑌵𑍋𑌤𑍍𑌤𑌮𑌂 𑌸𑌮𑍍𑌪𑍂𑌰𑍍𑌣𑌤𑌃 𑌉𑌤𑍍𑌕𑍃𑌷𑍍𑌟𑌂 𑌵𑌿𑌶𑍍𑌵𑌰𑍂𑌪𑌂 𑌵𑌿𑌶𑍍𑌵𑌮𑍇𑌵 𑌰𑍂𑌪𑌮𑌿𑌤𑌿 𑌕𑌰𑍍𑌮𑌧𑌾𑌰𑌯𑌃 । 𑌸𑌨𑍃𑌷𑍍𑌟𑌂 𑌸𑌰𑍍𑌵𑍋𑌤𑍍𑌤𑌮𑌵𑌿𑌶𑍍𑌵𑌰𑍂𑌪𑌂 𑌯𑌸𑍍𑌯 𑌸𑌃 𑌸𑌮𑍍𑌬𑍁𑌦𑍍𑌧𑍌 𑌸𑌨𑍍𑌦𑍃𑌷𑍍𑌟𑌸𑌰𑍍𑌵𑍋𑌤𑍍𑌤𑌮𑌵𑌿𑌶𑍍𑌵𑌰𑍂𑌪 । 𑌸𑍍𑌵𑌤𑍈𑌲𑌸𑌂𑌸𑍇𑌵𑌕𑌰𑍋𑌗𑌹𑌰𑍍𑌤𑍍𑌰𑍇 । 𑌪𑍍𑌰𑌸𑌿𑌦𑍍𑌧𑌂 𑌯𑌤𑍍 𑌕𑌵𑌿𑌃 𑌶𑍍𑌰𑍀𑌮𑌾𑌨𑌨𑍍𑌤𑌰𑌾𑌮𑌦𑍀𑌕𑍍𑌷𑌿𑌤𑌃 𑌸𑍍𑌵𑌯𑌂 𑌕𑍁𑌷𑍍𑌠𑍇𑌨 𑌰𑍁𑌗𑍍𑌣𑍋𑌪𑌿 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑌲𑍇𑌪𑌨𑌤𑍈𑌲𑌬𑌲𑌾𑌨𑍍𑌨𑍀𑌰𑍋𑌗𑌃 𑌸𑌮𑍍𑌪𑌨𑍍𑌨 𑌇𑌤𑌿 । 𑌤𑌂 𑌵𑍃𑌤𑍍𑌤𑌾𑌨𑍍𑌤𑌮𑍁𑌪𑌵𑌰𑍍𑌣𑌯𑌨𑍍𑌨𑌾𑌹 । 𑌸𑍍𑌵𑌸𑍍𑌯 𑌆𑌤𑍍𑌮𑌨𑌃 𑌤𑍈𑌲𑌮𑌭𑍍𑌯𑌞𑍍𑌜𑌨𑌂 𑌤𑌸𑍍𑌯 𑌸𑌂𑌸𑍇𑌵𑌕𑌾𑌨𑌾𑌮𑍁𑌪𑌯𑍋𑌕𑍍𑌤𑍄𑌣𑌾𑌂 𑌰𑍋𑌗𑌸𑍍𑌯 𑌕𑍁𑌷𑍍𑌠𑌾𑌦𑌿𑌕𑌸𑍍𑌯 𑌹𑌰𑍍𑌤𑌾 𑌵𑌿𑌨𑌾𑌶𑌯𑌿𑌤𑌾 𑌤𑌸𑍍𑌮𑍈 । 𑌹𑌰𑍍𑌤𑍁𑌸𑍍𑌤𑍃𑌨𑍍𑌨𑌨𑍍𑌤𑌤𑍍𑌵𑍇 𑌨 𑌲𑍋𑌕𑌾𑌵𑍍𑌯𑌵𑌨𑌿𑌷𑍍𑌠𑌾𑌖𑌲𑌰𑍍𑌥𑌤𑍃𑌨𑌾𑌮𑌿𑌤𑍍𑌯𑌸𑍍𑌮𑌾𑌤𑍍 𑌷𑌷𑍍𑌠𑍀𑌨𑌿𑌷𑍇𑌧𑌾𑌤𑍍 𑌦𑍍𑌵𑌿𑌤𑍀𑌯𑌾𑌯𑌾𑌮𑍇𑌵 𑌸𑌮𑍍𑌪𑌦𑍍𑌯𑌮𑌾𑌨𑌾𑌯𑌾𑌂 𑌤𑌤𑍍𑌪𑍁𑌰𑍁𑌷𑌸𑌮𑌾𑌸𑌦𑍌𑌰𑍍𑌲𑌭𑍍𑌯𑌾𑌤𑍍 𑌤𑌤𑍍𑌸𑍂𑌤𑍍𑌰𑌸𑍍𑌥𑌪𑍍𑌰𑌤𑍍𑌯𑌾𑌹𑌾𑌰𑍇 𑌤𑍃𑌚𑍋𑌭𑌾𑌵𑍇𑌨 𑌪𑍍𑌰𑌾𑌪𑍍𑌤𑌾 𑌤𑍃𑌜𑌨𑍍𑌤𑌤𑍍𑌵𑌨𑌿𑌷𑍍𑌪𑌤𑍍𑌤𑌿𑌰𑌾𑌶𑍍𑌰𑌯𑌣𑍀𑌯𑌾 । 𑌪𑍁𑌨𑌶𑍍𑌚 𑌵𑌾𑌤𑌾𑌲𑌯𑌾𑌧𑍀𑌶𑍇𑌤𑌿 𑌸𑌮𑍍𑌬𑍁𑌦𑍍𑌧𑍌 𑌨𑌮𑍋 𑌨𑌮𑌃 𑌤𑍇 𑌤𑍁𑌭𑍍𑌯𑌮𑍍 ।

    𑌬𑌾𑌲𑌾𑌨𑍍𑌸𑍍𑌵𑌕𑍀𑌯𑌾𑌂𑌸𑍍𑌤𑌵 𑌸𑌨𑍍𑌨𑌿𑌧𑌾𑌨𑍇 𑌦𑌿𑌵𑍍𑌯𑌾𑌨𑍍𑌨𑌦𑌾𑌨𑌾𑌤𑍍𑌪𑌰𑌿𑌪𑌾𑌲𑌯𑌦𑍍𑌭𑌿𑌃 ।
    𑌸𑌦𑌾 𑌪𑌠𑌦𑍍𑌭𑌿𑌶𑍍𑌚 𑌪𑍁𑌰𑌾𑌣𑌰𑌤𑍍𑌨𑌂 𑌸𑌂𑌸𑍇𑌵𑌿𑌤𑌾𑌯𑌾𑌸𑍍𑌤𑍁 𑌨𑌮𑍋 𑌹𑌰𑍇 𑌤𑍇 ॥ ௪ ॥

𑌬𑌾𑌲𑌾𑌨𑍍𑌪𑍁𑌤𑍍𑌤𑍍𑌰𑌕𑌾𑌨𑍍 𑌸𑍍𑌵𑌕𑍀𑌯𑌾𑌨𑍍𑌸𑍍𑌵𑍀𑌯𑌾𑌨𑍍 𑌪𑌰𑌿𑌪𑌾𑌲𑌯𑌦𑍍𑌭𑌿𑌰𑌿𑌤𑍍𑌯𑌨𑍍𑌵𑌯𑌃 । 𑌤𑌵 𑌭𑌵𑌤𑌃 𑌰𑍁𑌁𑌤𑍍𑌵𑌪𑍍𑌰𑌕𑌰𑌣𑌸𑍂𑌤𑍍𑌰𑍇𑌭𑍍𑌯𑍋𑌨𑍁𑌸𑍍𑌵𑌾𑌰𑌸𑌕𑌾𑌰𑍌 । 𑌸𑌨𑍍𑌨𑌿𑌧𑌾𑌨𑍇 𑌨𑌿𑌕𑌟𑍇 । 𑌦𑌿𑌵𑍍𑌯𑌾𑌨𑍍𑌨𑌦𑌾𑌨𑌾𑌤𑍍 𑌦𑌿𑌵𑍍𑌯𑌸𑍍𑌯 𑌦𑍇𑌵𑌸𑌮𑍍𑌬𑌦𑍍𑌧𑌸𑍍𑌯 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑌾𑌯 𑌨𑌿𑌵𑍇𑌦𑌿𑌤𑌤𑍍𑌵𑌾𑌤𑍍𑌤𑌤𑍍𑌪𑍍𑌰𑌸𑌾𑌦𑌭𑍂𑌤𑌸𑍍𑌯𑌾𑌨𑍍𑌨𑌸𑍍𑌯 𑌸𑍍𑌵𑌿𑌨𑍍𑌨𑌤𑌣𑍍𑌡𑍁𑌲𑌸𑍍𑌯 𑌦𑌾𑌨𑌾𑌦𑍍𑌵𑌿𑌤𑌰𑌣𑌾𑌤𑍍 𑌪𑌰𑌿𑌪𑌾𑌲𑌯𑌦𑍍𑌭𑌿𑌰𑌿𑌤𑍍𑌯𑌨𑍍𑌵𑌯𑌃 । 𑌪𑌰𑌿𑌪𑌾𑌲𑌯𑌦𑍍𑌭𑌿𑌃 𑌅𑌨𑍍𑌨𑌪𑍍𑌰𑌾𑌶𑌨𑌾𑌦𑌿𑌕𑌰𑍍𑌮𑌸𑍁 𑌪𑍋𑌷𑌯𑌦𑍍𑌭𑌿𑌃 𑌸𑌂𑌸𑍇𑌵𑌿𑌤𑌾𑌯𑍇𑌤𑍍𑌯𑌨𑍍𑌵𑌯𑌃 । 𑌸𑌦𑌾 𑌸𑌰𑍍𑌵𑌕𑌾𑌲𑍇𑌷𑍁 𑌪𑌠𑌦𑍍𑌬𑌿𑌃 𑌪𑌾𑌰𑌾𑌯𑌣𑌂 𑌕𑍁𑌰𑍍𑌵𑌦𑍍𑌬𑌿𑌃 𑌪𑍁𑌰𑌾𑌣𑌰𑌤𑍍𑌨𑌂 𑌶𑍍𑌰𑍀𑌮𑌦𑍍𑌭𑌾𑌗𑌵𑌤𑌂 𑌵𑌿𑌶𑌿𑌷𑍍𑌯 𑌦𑌶𑌮𑌸𑍍𑌕𑌨𑍍𑌧𑍇 𑌵𑌿𑌦𑍍𑌯𑌮𑌾𑌨𑌂 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑌚𑌰𑌿𑌤𑌮𑍍 । 𑌤𑌾𑌦𑍃𑌶𑍈𑌰𑍍𑌭𑌕𑍍𑌤𑍈𑌃 𑌸𑌂𑌸𑍇𑌵𑌿𑌤𑌾𑌯 𑌅𑌰𑍍𑌚𑌿𑌤𑌾𑌯 𑌅𑌸𑍍𑌤𑍁 𑌭𑌵𑌤𑍁 𑌨𑌮𑌃 𑌸𑌨𑍍𑌨𑌤𑌿𑌃 । 𑌹𑌰𑍇 𑌹𑌰𑌤𑌿 𑌪𑌾𑌪𑌾𑌨𑌿𑌤𑌿 𑌹𑌰𑌿𑌃 𑌸𑌮𑍍𑌬𑍁𑌦𑍍𑌧𑍌 𑌤𑍇 𑌤𑍁𑌭𑍍𑌯𑌮𑍍 ।

    𑌨𑌿𑌤𑍍𑌯𑌾𑌨𑍍𑌨𑌦𑌾𑌤𑍍𑌰𑍇 𑌚 𑌮𑌹𑍀𑌸𑍁𑌰𑍇𑌭𑍍𑌯𑍋 𑌨𑌿𑌤𑍍𑌯𑌂 𑌦𑌿𑌵𑌿𑌷𑍍𑌠𑍈𑌰𑍍𑌨𑌿𑌶𑌿 𑌪𑍂𑌜𑌿𑌤𑌾𑌯 ।
    𑌮𑌾𑌤𑍍𑌰𑌾 𑌚 𑌪𑌿𑌤𑍍𑌰𑌾 𑌚 𑌤𑌥𑍋𑌦𑍍𑌧𑌵𑍇𑌨 𑌸𑌮𑍍𑌪𑍂𑌜𑌿𑌤𑌾𑌯𑌾𑌸𑍍𑌤𑍁 𑌨𑌮𑍋 𑌨𑌮𑌸𑍍𑌤𑍇 ॥ ௫ ॥

𑌨𑌿𑌤𑍍𑌯𑌂 𑌸𑌦𑌾𑌨𑍍𑌨𑌂 𑌭𑍋𑌜𑌨𑌂 𑌤𑌸𑍍𑌯 𑌦𑌾𑌤𑌾 𑌵𑌿𑌤𑌰𑌣𑌸𑍍𑌤𑌸𑍍𑌮𑍈 𑌨𑌮𑍋 𑌨𑌮 𑌇𑌤𑍍𑌯𑌨𑍍𑌵𑌯𑌃 । 𑌮𑌹𑍀𑌸𑍁𑌰𑍇𑌭𑍍𑌯𑍋 𑌮𑌹𑍀 𑌭𑍂𑌃 𑌮𑌹𑍍𑌯𑌤 𑌇𑌤𑌿 । 𑌤𑌸𑍍𑌯𑌾𑌃 𑌸𑍁𑌰𑌾𑌃 𑌦𑍇𑌵𑌾𑌃 𑌇𑌵 𑌭𑍂𑌸𑍁𑌰𑌾 𑌬𑍍𑌰𑌾𑌹𑍍𑌮𑌣𑌾𑌃 𑌤𑍇𑌭𑍍𑌯𑍋 𑌦𑌾𑌤𑍍𑌰 𑌇𑌤𑍍𑌯𑌨𑍍𑌵𑌯𑌃 । 𑌨𑌿𑌤𑍍𑌯𑌮𑌨𑍁𑌦𑌿𑌨𑌂 । 𑌦𑌿𑌵𑌿𑌷𑍍𑌠𑍈𑌃 𑌦𑌿𑌵𑌿 𑌸𑍍𑌵𑌰𑍍𑌗𑍇 𑌤𑌿𑌷𑍍𑌠𑌤𑍀𑌤𑌿 𑌅𑌲𑍁𑌗𑍁𑌪𑌪𑌦𑌸𑌮𑌾𑌸𑌾𑌤𑍍𑌸𑌿𑌦𑍍𑌧𑌸𑍍𑌯 𑌅𑌮𑍍𑌬𑌾𑌮𑍍𑌬𑌗𑍋𑌭𑍂𑌮𑌿𑌸𑌵𑍍𑌯𑌾𑌪𑌦𑍍𑌵𑌿𑌤𑍍𑌰𑌿𑌕𑍁𑌶𑍇𑌕𑍁𑌶𑌙𑍍𑌕𑍍𑌵𑌙𑍍𑌗𑍁𑌮𑌞𑍍𑌜𑌿𑌪𑍁𑌞𑍍𑌜𑌿𑌪𑌰𑌮𑍇𑌬𑌰𑍍𑌹𑌿𑌰𑍍𑌦𑌿𑌵𑍍𑌯𑌗𑍍𑌨𑌿𑌭𑍍𑌯𑌃 𑌸𑍍𑌥 𑌇𑌤𑌿 𑌷𑌤𑍍𑌵𑌾𑌤𑍍 𑌷𑍍𑌟𑍁𑌤𑍍𑌵𑌾𑌚𑍍𑌚 𑌦𑌿𑌵𑌿𑌷𑍍𑌠𑍋𑌮𑌰𑌸𑍍𑌤𑍈𑌃 । 𑌨𑌿𑌶𑌿 𑌨𑌿𑌤𑌰𑌾𑌂 𑌶𑍍𑌯𑌤𑌿 𑌤𑌨𑍂𑌕𑌰𑍋𑌤𑌿 𑌵𑍍𑌯𑌾𑌪𑌾𑌰𑌾𑌨𑌿𑌤𑌿 𑌪𑍃𑌷𑍋𑌦𑌰𑌾𑌦𑌿𑌤𑍍𑌵𑌾𑌤𑍍 𑌨𑌿𑌡𑍍𑌰𑌾𑌤𑍍𑌰𑌿𑌸𑍍𑌤𑌸𑍍𑌯𑌾𑌮𑍍 । 𑌪𑍂𑌜𑌿𑌤𑌾𑌯𑌾𑌰𑍍𑌚𑌿𑌤𑌸𑍍𑌤𑌸𑍍𑌮𑍈 𑌨𑌮𑍋 𑌨𑌮 𑌇𑌤𑍍𑌯𑌨𑍍𑌵𑌯𑌃 । 𑌪𑍍𑌰𑌤𑌿𑌰𑌾𑌤𑍍𑌰𑌂 𑌸𑍁𑌰𑌾 𑌮𑌨𑍍𑌦𑌿𑌰𑌾𑌮𑍍 𑌆𑌶𑍍𑌰𑌿𑌤𑍍𑌯 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑌂 𑌪𑍂𑌜𑌯𑌨𑍍𑌤𑍀𑌤𑌿 । 𑌮𑌾𑌤𑍍𑌰𑌾 𑌜𑌨𑌨𑍍𑌯𑌾 𑌦𑍇𑌵𑌕𑍍𑌯𑌾 𑌚 𑌪𑌿𑌤𑍍𑌰𑌾 𑌜𑌨𑌕𑍇𑌨 𑌵𑌸𑍁𑌦𑍇𑌵𑍇𑌨 𑌚 𑌤𑌥𑌾 𑌤𑌦𑍍𑌵𑌦𑍇𑌵 𑌉𑌦𑍍𑌧𑌵𑍇𑌨 𑌤𑌦𑌭𑌿𑌧𑍇𑌨 𑌬𑌾𑌨𑍍𑌧𑌵𑍇𑌨 𑌸𑌮𑍍𑌪𑍂𑌜𑌿𑌤𑌾𑌯 𑌅𑌰𑍍𑌚𑌿𑌤𑌸𑍍𑌤𑌮𑍈 𑌪𑍁𑌨𑌶𑍍𑌚 𑌨𑌮𑍋 𑌨𑌮 𑌇𑌤𑍍𑌯𑌨𑍍𑌵𑌯𑌃 । 𑌦𑍇𑌵𑌕𑍀𑌵𑌸𑍁𑌦𑍇𑌵𑍋𑌦𑍍𑌧𑌵𑍈𑌃 𑌸𑌮𑍍𑌪𑍂𑌜𑌿𑌤𑍇 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑍇 𑌵𑍈𑌕𑍁𑌣𑍍𑌠𑌾𑌯 𑌪𑍍𑌰𑌤𑌿𑌨𑌿𑌵𑍃𑌤𑍍𑌤𑍇 𑌚 𑌤𑌸𑍍𑌮𑌿𑌨𑍍 𑌦𑍍𑌵𑌾𑌰𑌕𑌾𑌪𑍁𑌰𑍍𑌯𑌾𑌂 𑌸𑌮𑍁𑌦𑍍𑌰𑌮𑌗𑍍𑌨𑌾𑌯𑌾𑌮𑌥 𑌤𑌸𑍍𑌯 𑌦𑌿𑌵𑍍𑌯𑌾𑌯𑌾𑌂 𑌮𑍂𑌰𑍍𑌤𑍌 𑌸𑌮𑍁𑌦𑍍𑌰𑌤𑌲 𑌉𑌤𑍍𑌪𑍍𑌲𑌵𑌮𑌾𑌨𑌾𑌯𑌾𑌂 𑌦𑍃𑌷𑍍𑌟𑌾𑌯𑌾𑌂 𑌤𑌾𑌂 𑌗𑍁𑌰𑍁𑌵𑌾𑌯𑍂 𑌦𑍇𑌶𑌮𑌿𑌮𑌮𑌾𑌨𑍀𑌯 𑌪𑍍𑌰𑌤𑌿𑌷𑍍𑌠𑌾𑌪𑌯𑌾𑌮𑌾𑌸𑌤𑍁𑌰𑌿𑌤𑌿 𑌲𑍋𑌕𑌵𑌿𑌖𑍍𑌯𑌾𑌤𑌮𑍍 । 𑌅𑌸𑍍𑌤𑍁 𑌭𑌵𑌤𑍁 𑌨𑌮𑍋 𑌨𑌮𑌃 𑌤𑍇 𑌤𑍁𑌭𑍍𑌯𑌮𑍍 ।

    𑌅𑌨𑌨𑍍𑌤𑌰𑌾𑌮𑌾𑌖𑍍𑌯𑌮𑌖𑌿𑌪𑍍𑌰𑌣𑍀𑌤𑌂 𑌸𑍍𑌤𑍋𑌤𑍍𑌰𑌂 𑌪𑌠𑍇𑌦𑍍𑌯𑌸𑍍𑌤𑍁 𑌨𑌰𑌸𑍍𑌤𑍍𑌰𑌿𑌕𑌾𑌲𑌮𑍍 ।
    𑌵𑌾𑌤𑌾𑌲𑌯𑍇𑌶𑌸𑍍𑌯 𑌕𑍃𑌪𑌾𑌬𑌲𑍇𑌨 𑌲𑌭𑍇𑌤 𑌸𑌰𑍍𑌵𑌾𑌣𑌿 𑌚 𑌮𑌙𑍍𑌗𑌳𑌾𑌨𑌿 ॥ ௬ ॥

𑌫𑌲𑌶𑍍𑌰𑍁𑌤𑌿𑌰𑌿𑌯𑌮𑍍 । 𑌅𑌨𑌨𑍍𑌤𑌰𑌾𑌮𑌾𑌖𑍍𑌯𑌮𑌖𑌿𑌪𑍍𑌰𑌣𑍀𑌤𑌂 𑌅𑌨𑌨𑍍𑌤𑌰𑌾𑌮𑌾𑌖𑍍𑌯𑌃 𑌤𑌨𑍍𑌨𑌾𑌮𑌧𑍇𑌯𑌃 𑌕𑌵𑌿𑌃 𑌮𑌖𑍀 𑌦𑍀𑌕𑍍𑌷𑌿𑌤𑌶𑍍𑌚 𑌤𑍇𑌨 𑌪𑍍𑌰𑌣𑍀𑌤𑌂 𑌵𑌿𑌰𑌚𑌿𑌤𑌂 𑌸𑍍𑌤𑍋𑌤𑍍𑌰𑌂 𑌸𑍍𑌤𑌵𑌂 𑌪𑌠𑍇𑌦𑍍𑌬𑍍𑌰𑍂𑌯𑌾𑌤𑍍 𑌯𑌃 𑌯𑌕𑌃 𑌤𑍁 𑌨𑌰𑌃 𑌮𑌾𑌨𑌵𑌜𑌨𑍍𑌮𑌾 𑌤𑍍𑌰𑌿𑌕𑌾𑌲𑌂 𑌪𑍍𑌰𑌤𑌿𑌦𑌿𑌨𑌂 𑌤𑍍𑌰𑌿𑌃 । 𑌸 𑌇𑌤𑍍𑌯𑌧𑍍𑌯𑌾𑌹𑌾𑌰𑌃 𑌪𑌾𑌤𑌾𑌲𑌯𑍇𑌶𑌸𑍍𑌯 𑌮𑌨𑍍𑌦𑌿𑌰𑌸𑍍𑌯𑌾𑌸𑍍𑌯𑍇𑌶𑌸𑍍𑌯 𑌸𑍍𑌵𑌾𑌮𑌿𑌨𑌃 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑌸𑍍𑌯 𑌕𑍃𑌪𑌾𑌬𑌲𑍇𑌨 𑌕𑌰𑍁𑌣𑌾𑌪𑍍𑌰𑌭𑌾𑌵𑍇𑌣 𑌲𑌭𑍇𑌤 𑌸𑌮𑌵𑌾𑌪𑍍𑌨𑍁𑌯𑌾𑌤𑍍 𑌸𑌰𑍍𑌵𑌾𑌣𑌿 𑌸𑌕𑌲𑌾𑌨𑌿 𑌚 𑌮𑌙𑍍𑌗𑌳𑌾𑌨𑌿 𑌭𑌵𑍍𑌯𑌾𑌨𑌿 । 𑌏𑌵𑌂 𑌕𑌲𑍍𑌯𑌾𑌣𑌶𑌬𑍍𑌦𑌾𑌦𑌾𑌰𑌭𑍍𑌯 𑌮𑌙𑍍𑌗𑌳𑌶𑌬𑍍𑌦𑌾𑌤𑍍𑌸𑍍𑌤𑍁𑌤𑌿𑌮𑍁𑌪𑌸𑌂𑌹𑌰𑌤𑍀𑌤𑌿 𑌵𑌿𑌶𑍇𑌷𑌃 ।

    𑌗𑍁𑌰𑍁𑌵𑌾𑌤𑌪𑍁𑌰𑍀𑌶𑌪𑌞𑍍𑌚𑌕𑌾𑌖𑍍𑌯𑌂 𑌸𑍍𑌤𑍁𑌤𑌿𑌰𑌤𑍍𑌨𑌂 𑌪𑌠𑌤𑌾𑌂 𑌸𑍁𑌮𑌙𑍍𑌗𑌳𑌂 𑌸𑍍𑌯𑌾𑌤𑍍 ।
    𑌹𑍃𑌦𑌿 𑌚𑌾𑌪𑌿 𑌵𑌿𑌶𑍇𑌦𑍍𑌧𑌰𑌿𑌃 𑌸𑍍𑌵𑌯𑌂 𑌤𑍁 𑌰𑌤𑌿𑌨𑌾𑌥𑌾𑌯𑍁𑌤𑌤𑍁𑌲𑍍𑌯𑌦𑍇𑌹𑌕𑌾𑌨𑍍𑌤𑌿𑌃 ॥ ௭ ॥

𑌮𑌙𑍍𑌗𑌳𑌶𑍍𑌲𑍋𑌕𑍋𑌯𑌮𑍍 𑌔𑌪𑌚𑍍𑌛𑌨𑍍𑌦𑌸𑌿𑌕𑌵𑍃𑌤𑍍𑌤𑍇𑌨 𑌨𑌿𑌬𑌦𑍍𑌧𑌃 । 𑌗𑍁𑌰𑍁𑌵𑌾𑌤𑌪𑍁𑌰𑍀𑌶𑌪𑌞𑍍𑌚𑌕𑌾𑌖𑍍𑌯𑌂 𑌗𑍁𑌰𑍁𑌵𑌾𑌤𑌪𑍁𑌰𑍀 𑌗𑍁𑌰𑍁𑌵𑌾𑌯𑍁𑌪𑍁𑌰𑍀 𑌤𑌸𑍍𑌯𑌾 𑌈𑌶𑌃 𑌤𑌤𑍍𑌸𑍍𑌥𑌮𑌨𑍍𑌦𑌿𑌰𑌦𑍇𑌵𑌤𑌾 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑌃 । 𑌤𑌸𑍍𑌯 𑌪𑌞𑍍𑌚𑌕𑌂 𑌗𑍁𑌰𑍁𑌵𑌾𑌤𑌪𑍁𑌰𑍀𑌶𑌪𑌞𑍍𑌚𑌕𑌮𑍍 𑌆𑌖𑍍𑌯𑌾 𑌯𑌸𑍍𑌯 𑌤𑌤𑍍 𑌗𑍁𑌰𑍁𑌵𑌾𑌤𑌪𑍁𑌰𑍀𑌶𑌪𑌞𑍍𑌚𑌕𑌾𑌖𑍍𑌯𑌂 । 𑌤𑌤𑍍 𑌸𑍍𑌤𑍁𑌤𑌿𑌰𑌤𑍍𑌨𑌂 𑌸𑍍𑌤𑍁𑌤𑍀 𑌰𑌤𑍍𑌨𑌮𑌿𑌵 𑌶𑍍𑌰𑍇𑌷𑍍𑌠𑌂 𑌸𑍍𑌤𑍋𑌤𑍍𑌰𑌮𑌿𑌤𑌿 𑌯𑌾𑌵𑌤𑍍 । 𑌤𑌦𑌭𑌿𑌧𑌂 𑌸𑍍𑌤𑍁𑌤𑌿𑌰𑌤𑍍𑌨𑌂 𑌪𑌠𑌤𑌾𑌂 𑌵𑌦𑌤𑌾𑌂 𑌸𑍁𑌮𑌙𑍍𑌗𑌳𑌂 𑌸𑍁𑌷𑍍𑌠𑍁 𑌇𑌷𑍍𑌟𑌸𑌿𑌦𑍍𑌧𑌿𑌃 𑌸𑍍𑌯𑌾𑌤𑍍 𑌵𑌿𑌧𑍀𑌯𑌤𑍇 । 𑌹𑍃𑌦𑌿 𑌮𑌾𑌨𑌸𑍇 𑌚𑌾𑌪𑌿 𑌨 𑌕𑍇𑌵𑌲𑌮𑌿𑌷𑍍𑌟𑌸𑌿𑌦𑍍𑌧𑌿𑌲𑌾𑌭 𑌇𑌤𑍍𑌯𑌾𑌹 । 𑌵𑌿𑌶𑍇𑌤𑍍 𑌅𑌨𑍍𑌤𑌰𑍍𑌵𑌿𑌗𑌾𑌹𑌤𑍇 𑌹𑌰𑌿𑌃 𑌶𑍍𑌰𑍀𑌮𑌨𑍍𑌨𑌾𑌰𑌾𑌯𑌣𑌃 । 𑌝𑌯𑍋 𑌹𑍋𑌨𑍍𑌯𑌤𑌰𑌸𑍍𑌯𑌾𑌮𑌿𑌤𑌿 𑌵𑍈𑌕𑌲𑍍𑌪𑌿𑌕𑍋 𑌧𑌕𑌾𑌰𑌾𑌦𑍇𑌶𑌃 । 𑌸𑍍𑌵𑌯𑌂 𑌪𑌾𑌠𑌕𑌃 𑌤𑍁 𑌇𑌤𑍋𑌪𑌿 𑌫𑌲𑌂 𑌵𑌰𑍍𑌣𑌯𑌨𑍍𑌨𑌾𑌹 । 𑌰𑌤𑌿𑌨𑌾𑌥𑌾𑌯𑍁𑌤𑌤𑍁𑌲𑍍𑌯𑌦𑍇𑌹𑌕𑌾𑌨𑍍𑌤𑌿𑌃 𑌸𑌮𑍍𑌪𑌦𑍍𑌯𑍇𑌤𑍇𑌤𑌿 । 𑌰𑌤𑌿𑌃 𑌮𑌨𑍍𑌮𑌥𑌪𑌤𑍍𑌨𑍀 𑌤𑌸𑍍𑌯𑌾 𑌨𑌾𑌥𑌃 𑌪𑌤𑌿𑌃 𑌪𑍁𑌨𑌰𑍍𑌮𑌨𑍍𑌮𑌥 𑌏𑌵 । 𑌤𑌸𑍍𑌯 𑌅𑌯𑍁𑌤𑌾𑌨𑌿 𑌸𑌹𑌸𑍍𑌰𑌶𑌃 𑌤𑍈𑌃 𑌤𑍁𑌲𑍍𑌯𑌾 𑌤𑍋𑌲𑌨𑌯𑍋𑌗𑍍𑌯𑌾 𑌕𑌾𑌨𑍍𑌤𑌿𑌃 𑌶𑍋𑌭𑌾 𑌯𑌸𑍍𑌯 𑌤𑌾𑌦𑍃𑌶𑌃 𑌸𑍍𑌯𑌾𑌤𑍍 𑌸𑍍𑌵𑌯𑌂 𑌸𑍍𑌤𑍋𑌤𑍍𑌰𑌵𑌕𑍍𑌤𑍇𑌤𑌿 ।

    𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 ।
    𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 𑌨𑌾𑌰𑌾𑌯𑌣 ॥ ௯ ॥

𑌇𑌤𑍍𑌥𑌂 𑌤𑍁 𑌸𑍍𑌤𑍋𑌤𑍍𑌰𑌪𑌾𑌰𑌾𑌯𑌣𑌾𑌵𑌸𑌰𑍇 𑌪𑍍𑌰𑌤𑍍𑌯𑍇𑌕𑌂 𑌶𑍍𑌲𑍋𑌕𑌸𑍍𑌯 𑌪𑌶𑍍𑌚𑌾𑌜𑍍𑌜𑌪𑍍𑌯𑌤𑍇 । 𑌧𑍍𑌯𑌾𑌯𑌨𑍍𑌕𑍃𑌤𑍇 𑌯𑌜𑌨𑍍𑌯𑌜𑍍𑌞𑍈𑌸𑍍𑌤𑍍𑌰𑍇𑌤𑌾𑌯𑌾𑌂 𑌦𑍍𑌵𑌾𑌪𑌰𑍇𑌰𑍍𑌚𑌯𑌨𑍍 । 𑌯𑌦𑌾𑌪𑍍𑌨𑍋𑌤𑌿 𑌤𑌦𑌾𑌪𑍍𑌨𑍋𑌤𑌿 𑌕𑌲𑍌 𑌸𑌙𑍍𑌕𑍀𑌰𑍍𑌤𑍍𑌯 𑌕𑍇𑌶𑌵𑌮𑍍 ॥ 𑌇𑌤𑌿 𑌪𑍍𑌰𑌸𑌿𑌦𑍍𑌧𑌾𑌯𑌾𑌃 𑌶𑍍𑌰𑍀𑌮𑌦𑍍𑌭𑌾𑌗𑌵𑌤𑍋𑌕𑍍𑌤𑍇𑌰𑌨𑍁𑌗𑍁𑌣𑌂 𑌨𑌾𑌮𑌜𑌪𑌮𑌨𑍁𑌸𑌰𑍍𑌯𑌤𑍇 । 𑌨𑌾𑌰𑌾𑌯𑌣𑍇𑌤𑌿 𑌭𑌗𑌵𑌨𑍍𑌨𑌾𑌮 𑌨𑌾𑌨𑌾𑌰𑍍𑌥𑌮𑌤𑍍 । 𑌨𑌾𑌰𑌾 𑌜𑌲𑌮𑍍 𑌅𑌯𑌨𑌂 𑌸𑍍𑌥𑌾𑌨𑌂 𑌯𑌸𑍍𑌯𑍇𑌤𑌿 । 𑌯𑌦𑍍𑌵𑌾 𑌸𑌰𑍍𑌵𑍇 𑌗𑌤𑍍𑌯𑌰𑍍𑌥𑌾𑌃 𑌪𑍍𑌰𑌾𑌪𑍍𑌤𑍍𑌯𑌰𑍍𑌥𑌾 𑌇𑌤𑌿 𑌨𑌾𑌰𑌸𑍍𑌯 𑌜𑍍𑌞𑌾𑌨𑌸𑍍𑌯 𑌮𑍁𑌕𑍍𑌤𑍇𑌰𑍍𑌵𑌾 𑌅𑌯𑌨𑌂 𑌪𑍍𑌰𑌾𑌪𑍍𑌤𑌿𑌰𑍍𑌯𑌸𑍍𑌮𑌾𑌦𑌿𑌤𑌿 । 𑌯𑌦𑍍𑌵𑌾 𑌨𑌰𑌾𑌣𑌾𑌂 𑌸𑌮𑍂𑌹𑍋 𑌨𑌾𑌰𑌂 𑌤𑌤𑍍𑌰𑌾𑌯𑌨𑌂 𑌸𑍍𑌥𑌾𑌨𑌂 𑌯𑌸𑍍𑌯𑍇𑌤𑌿 । 𑌇𑌤𑌿 𑌶𑌤𑌶𑌃 𑌵𑍍𑌯𑍁𑌤𑍍𑌪𑌤𑍍𑌤𑌯𑌃 ।

॥ 𑌸𑌰𑍍𑌵𑌂 𑌶𑍍𑌰𑍀𑌕𑍃𑌷𑍍𑌣𑌾𑌰𑍍𑌪𑌣𑌮𑌸𑍍𑌤𑍁 ॥
